cmake_minimum_required(VERSION 3.24)
# Standalone CLion metadata only. Actual builds belong to b; no fetch or link.
if(APPLE)
    set(CMAKE_OSX_ARCHITECTURES arm64 CACHE STRING "Supported architecture")
    set(CMAKE_OSX_DEPLOYMENT_TARGET 14.0 CACHE STRING "Supported OS floor")
endif()
project(vexspoke_ide LANGUAGES C)
set(CMAKE_C_STANDARD 23)
set(CMAKE_C_STANDARD_REQUIRED YES)
set(CMAKE_C_EXTENSIONS YES)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
file(GLOB_RECURSE sources CONFIGURE_DEPENDS "src/*.c" "src/*.h")
if(APPLE)
    enable_language(OBJC)
    file(GLOB_RECURSE objc_sources CONFIGURE_DEPENDS "src/*.m")
    list(APPEND sources ${objc_sources})
endif()
list(FILTER sources EXCLUDE REGEX "/(attic|vendor|third_party|_old)/")
add_library(vexspoke_ide OBJECT EXCLUDE_FROM_ALL ${sources})
target_include_directories(vexspoke_ide PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/src")
# R2 split: canonical nio/io headers now live in the Relational Engine checkout.
# Optional local path keeps IDE navigation honest; missing headers stay real errors.
set(RELATIONAL_ENGINE_SOURCE_DIR "" CACHE PATH "Optional local Relational Engine src for canonical nio/io headers")
if(RELATIONAL_ENGINE_SOURCE_DIR)
    target_include_directories(vexspoke_ide PRIVATE "${RELATIONAL_ENGINE_SOURCE_DIR}")
endif()
find_path(VULKAN_INCLUDE_DIR vulkan/vulkan.h DOC "Optional local Vulkan headers; never downloaded")
if(VULKAN_INCLUDE_DIR)
    target_include_directories(vexspoke_ide PRIVATE "${VULKAN_INCLUDE_DIR}")
endif()
if(CMAKE_C_COMPILER_ID MATCHES "Clang|GNU")
    target_compile_options(vexspoke_ide PRIVATE -std=gnu23 -Wall -Wextra -Werror)
    if(APPLE)
        target_compile_options(vexspoke_ide PRIVATE -mcpu=apple-m1 "$<$<COMPILE_LANGUAGE:OBJC>:-fobjc-arc>")
    endif()
endif()
if(CMAKE_C_COMPILER_ID MATCHES "Clang")
    execute_process(COMMAND "${CMAKE_C_COMPILER}" -print-resource-dir
        OUTPUT_VARIABLE resource OUTPUT_STRIP_TRAILING_WHITESPACE)
    if(EXISTS "${resource}/include/stddef.h")
        target_compile_options(vexspoke_ide PRIVATE -isystem "${resource}/include")
    endif()
endif()
message(STATUS "IDE metadata only: excluded object target; use b for builds. No dependency downloads or linking.")
